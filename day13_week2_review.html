// ── main.jsx ──────────────────────────────────────────────
import { StrictMode, lazy, Suspense } from "react"
import { createRoot }  from "react-dom/client"
import { BrowserRouter } from "react-router-dom"
import { AuthProvider }  from "./contexts/AuthContext"
import { CartProvider }  from "./contexts/CartContext"
import { ThemeProvider } from "./contexts/ThemeContext"
import App               from "./App"
import "./index.css"

createRoot(document.getElementById("root")).render(
  <StrictMode>
    <BrowserRouter>
      <AuthProvider>
        <CartProvider>
          <ThemeProvider>
            <App />
          </ThemeProvider>
        </CartProvider>
      </AuthProvider>
    </BrowserRouter>
  </StrictMode>
)

// ── App.jsx ───────────────────────────────────────────────
const Products     = lazy(() => import("./pages/Products"))
const Dashboard    = lazy(() => import("./pages/Dashboard"))

function App() {
  return (
    <>
      <Navbar />
      <Suspense fallback={<Spinner />}>
        <Routes>
          {/* Public */}
          <Route path="/"                element={<Home />}          />
          <Route path="/products"        element={<Products />}      />
          <Route path="/products/:id"   element={<ProductDetail />} />
          <Route path="/login"          element={<Login />}         />

          {/* Protected — RequireAuth wrapper */}
          <Route element={<RequireAuth />}>
            <Route path="/cart"         element={<Cart />}          />
            <Route path="/dashboard"   element={<Dashboard />}     >
              <Route index              element={<DashboardHome />} />
              <Route path="orders"     element={<Orders />}       />
              <Route path="settings"   element={<Settings />}     />
            </Route>
          </Route>

          <Route path="*"              element={<NotFound />}       />
        </Routes>
      </Suspense>
    </>
  )
}

function isActive(linkTo, currentPath, end = false) {
  const to  = linkTo.replace(/\/$/, "")
  const cur = currentPath.replace(/\/$/, "")
  return end ? cur === to : cur === to || cur.startsWith(to + "/")
}

function generateBreadcrumbs(pathname) {
  const parts = pathname.split("/").filter(Boolean)
  return parts.map((seg, i) => ({
    label: seg[0].toUpperCase() + seg.slice(1),
    path:  "/" + parts.slice(0, i + 1).join("/")
  }))
}

function getFilteredProducts(products, { category, q, sort }) {
  return [...products]
    .filter(p => !category || category === "all" || p.category === category)
    .filter(p => !q         || p.name.toLowerCase().includes(q.toLowerCase()))
    .sort((a, b) => {
      if (sort === "price-asc")  return a.price  - b.price
      if (sort === "price-desc") return b.price  - a.price
      if (sort === "rating")     return b.rating - a.rating
      return a.name.localeCompare(b.name)
    })
}

function getCartStats(items) {
  return {
    itemCount: items.reduce((s,i) => s+i.qty, 0),
    total:     items.reduce((s,i) => s+i.price*i.qty, 0),
    isEmpty:   items.length === 0
  }
}

Custom hooks extraction — useFetch, useDebounce, useLocalStorage
Day 10
Extract three custom hooks that are reused across the app. useFetch(url) handles data fetching with AbortController. useDebounce(value, delay) delays the search param update. useLocalStorage(key, initial) syncs state to localStorage. Verify each hook in isolation.
Acceptance criteria
  • useFetch: returns { data, loading, error } + AbortController cleanup on url change
  • useFetch: race condition safe — rapid url changes don't show stale data
  • useDebounce: returns debounced value — only updates after delay ms of silence
  • useLocalStorage: initialises from storage, persists on every set, supports functional updates
  • All three hooks are in hooks/ folder and exported as named exports
