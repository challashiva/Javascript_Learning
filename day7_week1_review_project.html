javascriptTaskStore class — state management with private fields + closure
// ── Day 5 (Classes) + Day 1 (closures) + Day 2 (immutable updates) ──
class TaskStore {
  #tasks     = []          // private field — Day 5
  #listeners = []          // subscribers — Day 5 EventEmitter pattern
  #history   = []
  #nextId

  constructor() {
    this.#nextId = this.#makeIdGenerator()  // closure — Day 3
  }

  // Generator for unique IDs — Day 6
  *#makeIdGenerator() {
    let id = 1
    while (true) yield `task_${id++}`
  }

  add({ text, priority = "medium", category = "personal" }) {
    const task = {
      id:        this.#nextId.next().value,  // generator — Day 6
      text,
      priority,
      category,
      done:      false,
      createdAt: new Date()
    }
    this.#tasks = [...this.#tasks, task]  // immutable — Day 2
    this.#log(`Added: "${text}"`)
    this.#notify()
    return this
  }

  toggle(id) {
    this.#tasks = this.#tasks.map(t =>   // map — Day 2
      t.id === id ? { ...t, done: !t.done } : t
    )
    this.#notify()
  }

  delete(id) {
    const task = this.#tasks.find(t => t.id === id)
    this.#tasks = this.#tasks.filter(t => t.id !== id)  // filter — Day 2
    this.#log(`Deleted: "${task?.text}"`)
    this.#notify()
  }

  getFiltered({ filter = "all", search = "", sort = "priority" } = {}) {
    const PRIORITY = { high: 3, medium: 2, low: 1 }

    return [...this.#tasks]       // spread copy to avoid mutating — Day 2
      .filter(t => {               // filter + chaining — Day 2
        if (filter === "done")    return t.done
        if (filter === "pending") return !t.done
        if (filter === "high")    return t.priority === "high"
        return true
      })
      .filter(t => !search || t.text.toLowerCase().includes(search.toLowerCase()))
      .sort((a, b) => {
        if (sort === "priority") return PRIORITY[b.priority] - PRIORITY[a.priority]
        if (sort === "alpha")    return a.text.localeCompare(b.text)
        return b.createdAt - a.createdAt  // newest first
      })
  }

  getStats() {
    return this.#tasks.reduce((acc, t) => ({  // reduce — Day 2
      ...acc,
      total:   acc.total   + 1,
      done:    acc.done    + (t.done ? 1 : 0),
      pending: acc.pending + (t.done ? 0 : 1),
      high:    acc.high    + (t.priority === "high" && !t.done ? 1 : 0)
    }), { total:0, done:0, pending:0, high:0 })
  }

  // Subscribe to state changes — Day 5 pattern
  subscribe(fn) {
    this.#listeners.push(fn)
    return () => { this.#listeners = this.#listeners.filter(l => l !== fn) }
  }

  #notify() { this.#listeners.forEach(fn => fn()) }
  #log(msg)  { this.#history.unshift({ msg, time: new Date() }) }
  getHistory() { return this.#history.slice(0, 10) }
}

// ── Render function — data → DOM ───────────────────────────
// Day 5 (DOM manipulation) + Day 4 (template strings)
function renderTasks(tasks, container) {
  if (tasks.length === 0) {
    container.innerHTML = '<div class="tm-empty">No tasks here yet.</div>'
    return
  }
  // Use DocumentFragment for batch DOM insert — one reflow
  const frag = document.createDocumentFragment()

  tasks.forEach(task => {
    const el = document.createElement("div")
    el.className = `tm-item${task.done ? " done-item" : ""}`
    el.dataset.id = task.id

    el.innerHTML = `
      <div class="tm-check${task.done ? " checked" : ""}" data-action="toggle"></div>
      <span class="tm-text${task.done ? " striked" : ""}">${escapeHtml(task.text)}</span>
      <span class="tm-priority tm-pri-${task.priority}">${task.priority}</span>
      <div class="tm-actions">
        <button class="tm-btn-sm" data-action="delete">delete</button>
      </div>`
    frag.appendChild(el)
  })

  container.innerHTML = ""
  container.appendChild(frag)
}

// ── Event delegation on list container — Day 5 ─────────────
document.getElementById("tm-list").addEventListener("click", (e) => {
  const action = e.target.dataset.action
  const item   = e.target.closest(".tm-item")    // closest() — Day 5
  if (!item) return
  const { id } = item.dataset                    // destructuring — Day 2
  if (action === "toggle") store.toggle(id)
  if (action === "delete") store.delete(id)
})

// ── Debounced search — Day 3 (closure) ────────────────────
const debounce = (fn, ms) => {
  let timer
  return (...args) => { clearTimeout(timer); timer = setTimeout(() => fn(...args), ms) }
}
document.getElementById("tm-search")
  .addEventListener("input", debounce((e) => {
    state.search = e.target.value
    render()
  }, 200))
