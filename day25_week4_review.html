// prisma/schema.prisma
generator client { provider = "prisma-client-js" }

datasource db {
  provider  = "postgresql"
  url       = env("DATABASE_URL")
  directUrl = env("DIRECT_URL")  // for migrations (bypasses pooler)
}

model User {
  id        String    @id @default(cuid())
  clerkId   String    @unique
  email     String    @unique
  name      String
  imageUrl  String?
  role      Role      @default(USER)
  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  orders    Order[]
  reviews   Review[]
  cart      CartItem[]
  @@map("users")
}
enum Role { USER  ADMIN  MODERATOR }

model Category {
  id        String    @id @default(cuid())
  name      String    @unique
  slug      String    @unique
  imageUrl  String?
  products  Product[]
  @@map("categories")
}

model Product {
  id          String    @id @default(cuid())
  name        String
  slug        String    @unique
  description String
  price       Decimal   @db.Decimal(10,2)
  comparePriceDecimal?  @db.Decimal(10,2)  // original price for "sale"
  stock       Int       @default(0)
  images      String[]
  published   Boolean   @default(false)
  featured    Boolean   @default(false)
  rating      Float     @default(0)
  reviewCount Int       @default(0)
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt
  categoryId  String
  category    Category  @relation(fields:[categoryId],references:[id])
  reviews     Review[]
  orderItems  OrderItem[]
  cartItems   CartItem[]
  @@index([categoryId])
  @@index([published, featured])
  @@index([published, createdAt])
  @@map("products")
}

model Order {
  id          String       @id @default(cuid())
  status      OrderStatus  @default(PENDING)
  total       Decimal      @db.Decimal(10,2)
  address     Json?        // { line1, city, country, zip }
  createdAt   DateTime     @default(now())
  userId      String
  user        User         @relation(fields:[userId],references:[id],onDelete:CASCADE)
  items       OrderItem[]
  @@index([userId])
  @@map("orders")
}
enum OrderStatus { PENDING  CONFIRMED  SHIPPED  DELIVERED  CANCELLED }

model OrderItem {
  id        String   @id @default(cuid())
  quantity  Int
  price     Decimal  @db.Decimal(10,2)  // price at time of purchase
  orderId   String
  order     Order    @relation(fields:[orderId],references:[id],onDelete:CASCADE)
  productId String
  product   Product  @relation(fields:[productId],references:[id])
  @@map("order_items")
}

model Review {
  id        String   @id @default(cuid())
  rating    Int      // 1–5
  title     String?
  body      String
  createdAt DateTime @default(now())
  userId    String
  user      User     @relation(fields:[userId],references:[id])
  productId String
  product   Product  @relation(fields:[productId],references:[id])
  @@unique([userId, productId])  // one review per product per user
  @@index([productId])
  @@map("reviews")
}

model CartItem {
  id        String   @id @default(cuid())
  quantity  Int      @default(1)
  userId    String
  user      User     @relation(fields:[userId],references:[id],onDelete:CASCADE)
  productId String
  product   Product  @relation(fields:[productId],references:[id])
  @@unique([userId, productId])
  @@map("cart_items")
}
04 Auth layer — middleware, webhook, server patterns clerk

// middleware.ts — route protection
import { clerkMiddleware, createRouteMatcher } from "@clerk/nextjs/server"
import { NextResponse }                        from "next/server"

const isPublic = createRouteMatcher([
  "/", "/products(.*)", "/sign-in(.*)", "/sign-up(.*)",
  "/api/webhooks(.*)", "/api/products(.*)"  // public product browsing
])
const isAdmin = createRouteMatcher(["/admin(.*)", "/api/admin(.*)"])

export default clerkMiddleware(async (auth, req) => {
  if (!isPublic(req)) await auth.protect()

  if (isAdmin(req)) {
    const { sessionClaims } = await auth()
    if (sessionClaims?.metadata?.role !== "admin") {
      return NextResponse.redirect(new URL("/unauthorized", req.url))
    }
  }
})

// app/api/webhooks/clerk/route.ts — sync users to DB
import { Webhook } from "svix"
import { headers } from "next/headers"

export async function POST(req: Request) {
  const wh = new Webhook(process.env.WEBHOOK_SECRET!)
  const hdrs = await headers()
  const body = await req.text()

  let event
  try {
    event = wh.verify(body, {
      "svix-id":        hdrs.get("svix-id")!,
      "svix-timestamp": hdrs.get("svix-timestamp")!,
      "svix-signature": hdrs.get("svix-signature")!,
    })
  } catch { return new Response("Unauthorized", { status: 401 }) }

  const { type, data } = event as any
  const email = data.email_addresses?.[0]?.email_address ?? ""

  if (type === "user.created" || type === "user.updated") {
    await db.user.upsert({
      where:  { clerkId: data.id },
      update: { email, name: `${data.first_name} ${data.last_name}`.trim(), imageUrl: data.image_url },
      create: { clerkId: data.id, email, name: "" },
    })
  }
  if (type === "user.deleted") await db.user.delete({ where: { clerkId: data.id } })

  return Response.json({ ok: true })
}

// lib/auth.ts — server-side helpers used everywhere
import { auth } from "@clerk/nextjs/server"
import { redirect } from "next/navigation"

export async function requireAuth() {
  const { userId } = await auth()
  if (!userId) redirect("/sign-in")
  return userId
}

export async function requireAdmin() {
  const { userId, sessionClaims } = await auth()
  if (!userId) redirect("/sign-in")
  if (sessionClaims?.metadata?.role !== "admin") redirect("/unauthorized")
  return userId
}
05 API layer — Route Handlers + Server Actions api
typescriptactions/orders.ts — createOrder Server Action (atomic, auth-protected)
"use server"
import { auth }           from "@clerk/nextjs/server"
import { revalidatePath } from "next/cache"
import { db }             from "@/lib/db"
import { z }              from "zod"

const orderSchema = z.object({
  items: z.array(z.object({
    productId: z.string().cuid(),
    quantity:  z.number().int().positive(),
  })).min(1, "Order must have at least one item"),
  address: z.object({
    line1:   z.string().min(1),
    city:    z.string().min(1),
    country: z.string().length(2),
    zip:     z.string().min(3),
  }),
})

export async function createOrder(input: unknown) {
  const { userId } = await auth()
  if (!userId) return { error: "Unauthorized" }

  const parsed = orderSchema.safeParse(input)
  if (!parsed.success) return { error: "Invalid order", details: parsed.error.flatten() }

  const { items, address } = parsed.data

  try {
    const order = await db.$transaction(async (tx) => {
      // 1. Fetch and validate all products
      const products = await Promise.all(
        items.map(i => tx.product.findUniqueOrThrow({ where: { id: i.productId, published: true } }))
      )

      // 2. Check stock
      for (let idx = 0; idx < items.length; idx++) {
        if (products[idx].stock < items[idx].quantity)
          throw new Error(`Insufficient stock: ${products[idx].name}`)
      }

      // 3. Calculate total
      const total = items.reduce((s, item, i) =>
        s + Number(products[i].price) * item.quantity, 0)

      // 4. Create order + items + decrement stock (all atomic)
      const [dbUser] = await Promise.all([
        tx.user.findUniqueOrThrow({ where: { clerkId: userId } })
      ])

      const order = await tx.order.create({
        data: {
          userId: dbUser.id, total, address,
          items: { create: items.map((item, i) => ({
            productId: item.productId,
            quantity:  item.quantity,
            price:     products[i].price,
          })) }
        },
        include: { items: true }
      })

      await Promise.all(
        items.map((item) => tx.product.update({
          where: { id: item.productId },
          data:  { stock: { decrement: item.quantity } }
        }))
      )

      return order
    })

    revalidatePath("/dashboard/orders")
    return { success: true, orderId: order.id }
  } catch (e) {
    return { error: (e as Error).message || "Failed to create order" }
  }
}
